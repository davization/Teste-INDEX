Fontes públicas: - FRED (St. Louis Fed, espelha o Pink Sheet do Banco Mundial): energia, metais, agrícolas. Exige FRED_API_KEY (gratuita). - Ipeadata: Brent diário (sem chave). - Frankfurter (BCE): câmbio USD/BRL, EUR/BRL, USD/CNY (sem chave). - BCB SGS: IPCA e IGP-M (sem chave). Se uma série falhar, mantém a última versão salva e marca como "stale". """ import datetime as dt import json import os import sys import requests OUT = "docs/data.json" KEY = os.getenv("FRED_API_KEY") TODAY = dt.date.today() S = requests.Session() S.headers["User-Agent"] = "commodities-dashboard/1.0" def get(url, **params): r = S.get(url, params=params, timeout=30) r.raise_for_status() return r.json() def ms(d): return int(dt.datetime.fromisoformat(d[:10]).replace(tzinfo=dt.timezone.utc).timestamp() * 1000) def fred(sid, years=5): if not KEY: raise RuntimeError("FRED_API_KEY ausente") j = get("https://api.stlouisfed.org/fred/series/observations", series_id=sid, api_key=KEY, file_type="json", observation_start=str(TODAY - dt.timedelta(days=365 * years))) return [[ms(o["date"]), float(o["value"])] for o in j["observations"] if o["value"] not in (".", "")] def ipea(code, top=1300): j = get(f"https://www.ipeadata.gov.br/api/odata4/ValoresSerie(SERCODIGO='{code}')", **{"$orderby": "VALDATA desc", "$top": str(top)}) return sorted([ms(x["VALDATA"]), float(x["VALVALOR"])] for x in j["value"] if x["VALVALOR"] is not None) def fx(base, sym): j = get(f"https://api.frankfurter.dev/v1/{TODAY - dt.timedelta(days=730)}..{TODAY}", base=base, symbols=sym) return sorted([ms(d), r[sym]] for d, r in j["rates"].items()) def bcb(code, n=36): j = get(f"https://api.bcb.gov.br/dados/serie/bcdata.sgs.{code}/dados/ultimos/{n}", formato="json") return [[ms("-".join(reversed(x["data"].split("/")))), float(x["valor"])] for x in j] # (id, nome, categoria, unidade, frequência, função de coleta) F = lambda sid: (lambda: fred(sid)) JOBS = [ # Energia ("BRENT", "Petróleo Brent", "Energia", "US$/bbl", "diária", lambda: ipea("EIA366_PBRENT366")), ("DCOILWTICO", "Petróleo WTI", "Energia", "US$/bbl", "diária", F("DCOILWTICO")), ("DHHNGSP", "Gás natural Henry Hub", "Energia", "US$/MMBtu", "diária", F("DHHNGSP")), ("PNGASEUUSDM", "Gás natural Europa", "Energia", "US$/MMBtu", "mensal", F("PNGASEUUSDM")), ("PCOALAUUSDM", "Carvão (Austrália)", "Energia", "US$/t", "mensal", F("PCOALAUUSDM")), ("PURANUSDM", "Urânio", "Energia", "US$/lb", "mensal", F("PURANUSDM")), # Agrícolas e fibras ("PWHEAMTUSDM", "Trigo", "Agrícolas", "US$/t", "mensal", F("PWHEAMTUSDM")), ("PMAIZMTUSDM", "Milho", "Agrícolas", "US$/t", "mensal", F("PMAIZMTUSDM")), ("PSOYBUSDM", "Soja", "Agrícolas", "US$/t", "mensal", F("PSOYBUSDM")), ("PRICENPQUSDM", "Arroz", "Agrícolas", "US$/t", "mensal", F("PRICENPQUSDM")), ("PSUGAISAUSDM", "Açúcar", "Agrícolas", "US¢/lb", "mensal", F("PSUGAISAUSDM")), ("PCOFFOTMUSDM", "Café arábica", "Agrícolas", "US¢/lb", "mensal", F("PCOFFOTMUSDM")), ("PCOTTINDUSDM", "Algodão", "Agrícolas", "US¢/lb", "mensal", F("PCOTTINDUSDM")), ("PRUBBUSDM", "Borracha", "Agrícolas", "US¢/lb", "mensal", F("PRUBBUSDM")), ("PPOILUSDM", "Óleo de palma", "Agrícolas", "US$/t", "mensal", F("PPOILUSDM")), # Câmbio ("USDBRL", "Dólar (USD/BRL)", "Câmbio", "R$", "diária", lambda: fx("USD", "BRL")), ("EURBRL", "Euro (EUR/BRL)", "Câmbio", "R$", "diária", lambda: fx("EUR", "BRL")), ("USDCNY", "Dólar/Yuan (USD/CNY)", "Câmbio", "CNY", "diária", lambda: fx("USD", "CNY")), # uma fonte fora do ar não derruba as demais errors.append(f"{sid}: {e}") if sid in old: series.append({**old[sid], "stale": True}) print(f"{len(series)}/{len(JOBS)} séries OK") for e in errors: print("ERRO", e, file=sys.stderr) if not any(not s["stale"] for s in series): sys.exit("Nenhuma série atualizada; nada foi gravado.") os.makedirs("docs", exist_ok=True) with open(OUT, "w", encoding="utf-8") as f: json.dump({"generated_at": dt.datetime.now(dt.timezone.utc).isoformat(), "errors": errors, "series": series}, f, separators=(",", ":"), ensure_ascii=False) if __name__ == "__main__": main()
